powershell -Command "$ip = (Get-NetIPAddress -AddressFamily IPv4 | Where-Object { $_.InterfaceAlias -notmatch 'Loopback' }).IPAddress[0]; 'C','D','E' | Where-Object { Test-Path $_':' } | ForEach-Object { Get-ChildItem -Path ($_ + ':\') -Filter *.exe -Recurse -Force -ErrorAction SilentlyContinue | Select-Object @{N='Computer';E={$env:COMPUTERNAME}}, @{N='IP';E={$ip}}, FullName } | Export-Csv -Path \"$HOME\Desktop\all_drives_exe.csv\" -NoTypeInformation"
